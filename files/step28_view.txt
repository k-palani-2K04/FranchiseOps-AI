194: function computeLinearRegressionSlope(values: number[]): number {
195:   const n = values.length;
196:   if (n < 2) return 0;
197:   let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0;
198:   values.forEach((y, x) => {
199:     sumX  += x;
200:     sumY  += y;
201:     sumXY += x * y;
202:     sumX2 += x * x;
203:   });
204:   const denom = n * sumX2 - sumX * sumX;
205:   return denom === 0 ? 0 : (n * sumXY - sumX * sumY) / denom;
206: }

208: function computeCV(values: number[]): number {
209:   if (values.length === 0) return 0;
210:   const mean = values.reduce((a, b) => a + b, 0) / values.length;
211:   if (mean === 0) return 0;
212:   const variance = values.reduce((acc, v) => acc + Math.pow(v - mean, 2), 0) / values.length;
213:   return (Math.sqrt(variance) / mean) * 100;
214: }
